program -> declaration* EOF ;

declaration -> funDec | varDecl | classDecl | statement ;

classDecl -> "class" IDENTIFIER "{" (function | getter)* "}" ;

varDecl -> "var" IDENTIFIER ( "=" expression )? ";" ;
funDec -> "fun" function ;
parameters -> IDENTIFIER ( "," IDENTIFIER )* ;
function -> IDENTIFIER "(" parameters? ")" block ;
getter -> IDENTIFIER block ;

statement -> exprStmt | printStmt | ifStmt | whileStmt | forStmt | breakStmt | returnStmt | block;
breakStmt -> "break" ";" ;
ifStmt -> "if" "(" expression ")" statement ( "else" statement )? ;
whileStmt -> "while" "(" expression ")" statement  ;
forStmt -> "for" "(" ( varDecl | exprStmt | ";" ) expression? ";" expression? ")" statement ;
returnStmt -> "return" expression? ";" ;
exprStmt -> expression ";" ;
printStmt -> "print" expression ";" ;
block -> "{" declaration* "}" ;

expression -> comma ;

comma -> assignment ( ","  assignment )*  ;
assignment -> ( call "." )? IDENTIFIER "=" assignment | logic_or ;
logic_or -> logic_and ( "or" logic_and )* ;
logic_and -> ternary ( "and" ternary)* ;
ternary -> equality ( "?" expression ":" ternary )? ;
equality -> comparison ( ( "==" | "!=" ) comparison )* ;
comparison -> term ( ( ">" | ">=" | "<" | "<=" ) term )* ;
term -> factor ( ("+" | "-" ) factor )* ;
factor -> unary ( ( "/" | "*" ) unary )* ;
unary -> ("!" | "-") unary | call ;
arguments -> expression ( "," expression )* ;
call -> primary ( "(" arguments? ")" | "." IDENTIFIER )* ;
primary -> STRING | NUMBER | "true" | "false" | "nil" | "(" expression ")" | IDENTIFIER | lambda ;
lambda -> "fun" "(" parameters? ")" block;
